import pandas as pd
import numpy as np

btc_close = btc['Close'].copy()
if isinstance(btc_close, pd.Series):
    btc_close = btc_close.to_frame(name='BTC')
else:
    btc_close = btc_close.iloc[:, [0]] 
    btc_close.columns = ['BTC']

dxy_close = dxy['Close'].copy()
if isinstance(dxy_close, pd.Series):
    dxy_close = dxy_close.to_frame(name='DXY')
else:
    dxy_close = dxy_close.iloc[:, [0]]
    dxy_close.columns = ['DXY']

df_aligned = pd.merge(dxy_close, btc_close, left_index=True, right_index=True, how='inner')

df_aligned = df_aligned.dropna()

print(f"valid trading: {len(df_aligned)}")
print(df_aligned.head())

from sklearn.preprocessing import OneHotEncoder
from sklearn.preprocessing import MinMaxScaler

regimes = np.zeros(len(df_aligned))
for i, cp in enumerate(result):
    if i == 0:
        regimes[:cp] = 0
    else:
        regimes[result[i-1]:cp] = i

df_aligned['Regime_ID'] = regimes

days_since = np.zeros(len(df_aligned))
last_cp = 0
for i in range(len(df_aligned)):
    if i in result:
        last_cp = i
    days_since[i] = i - last_cp

df_aligned['Days_Since_CP'] = days_since

encoder = OneHotEncoder(sparse_output=False)
regime_encoded = encoder.fit_transform(df_aligned[['Regime_ID']])
regime_df = pd.DataFrame(regime_encoded, index=df_aligned.index, 
                         columns=[f'Regime_{int(i)}' for i in range(regime_encoded.shape[1])])

features = pd.concat([df_aligned[['BTC_Ret', 'DXY_Ret', 'Days_Since_CP']], regime_df], axis=1)

scaler = MinMaxScaler()
scaled_features = scaler.fit_transform(features)

# Windows for LSTM
def create_lstm_dataset(data, target_col_idx, time_steps=10):
    X, y = [], []
    for i in range(len(data) - time_steps):
        X.append(data[i:(i + time_steps), :])
        y.append(data[i + time_steps, target_col_idx])
    return np.array(X), np.array(y)

X_train, y_train = create_lstm_dataset(scaled_features, target_col_idx=0, time_steps=5)

print(f"inputshape (sample, step, feature): {X_train.shape}")
